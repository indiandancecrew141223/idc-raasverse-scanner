<script>


  // ======================================================

  // IMPORTANT:

  // REPLACE THIS WITH YOUR APPS SCRIPT /exec URL

  // ======================================================


  const APPS_SCRIPT_URL =

    'https://script.google.com/macros/s/AKfycbzKpcjtDtmo75CPtmWKxTkSQkw-gWdKc7fAlBUR8ru6lYBxh3MPBKknnTBJl9CKACtA/exec';




  let scanner = null;


  let scanLocked = false;




  const welcomeScreen =

    document.getElementById(

      'welcomeScreen'

    );



  const scannerScreen =

    document.getElementById(

      'scannerScreen'

    );



  const scanButton =

    document.getElementById(

      'scanButton'

    );



  const cancelButton =

    document.getElementById(

      'cancelButton'

    );



  const cameraError =

    document.getElementById(

      'cameraError'

    );



  const scanningMessage =

    document.getElementById(

      'scanningMessage'

    );



  const verifyMessage =

    document.getElementById(

      'verifyMessage'

    );




  // ======================================================

  // SCAN BUTTON

  // ======================================================


  scanButton.addEventListener(

    'click',

    startScanner

  );




  // ======================================================

  // CANCEL BUTTON

  // ======================================================


  cancelButton.addEventListener(

    'click',

    async function() {


      await stopScanner();


      scannerScreen

        .classList

        .add('hidden');


      welcomeScreen

        .classList

        .remove('hidden');


    }

  );




  // ======================================================

  // START CAMERA

  // ======================================================


  async function startScanner() {


    scanLocked = false;



    cameraError

      .classList

      .add('hidden');



    verifyMessage

      .classList

      .add('hidden');



    scanningMessage

      .classList

      .remove('hidden');



    welcomeScreen

      .classList

      .add('hidden');



    scannerScreen

      .classList

      .remove('hidden');



    try {



      scanner =

        new Html5Qrcode(

          'reader'

        );



      const cameras =

        await Html5Qrcode

          .getCameras();



      if (

        !cameras ||

        cameras.length === 0

      ) {


        throw new Error(

          'No camera was found on this device.'

        );


      }



      let selectedCamera =

        cameras[0].id;



      // Prefer rear camera


      for (

        let i = 0;

        i < cameras.length;

        i++

      ) {


        const label =

          String(

            cameras[i].label || ''

          )

          .toLowerCase();



        if (

          label.includes('back') ||

          label.includes('rear') ||

          label.includes('environment')

        ) {


          selectedCamera =

            cameras[i].id;


          break;


        }


      }



      const config = {


        fps: 12,


        qrbox: function(

          viewfinderWidth,

          viewfinderHeight

        ) {


          const size =

            Math.floor(

              Math.min(

                viewfinderWidth,

                viewfinderHeight

              ) * .72

            );



          return {


            width: size,


            height: size


          };


        },


        aspectRatio:

          1.0


      };



      await scanner.start(


        selectedCamera,


        config,


        onScanSuccess,


        function() {

          // Normal frame-by-frame

          // QR detection failures ignored.

        }


      );



    }

    catch (error) {



      console.error(

        error

      );



      cameraError

        .textContent =

        'Camera could not be opened. Please allow camera access in your browser settings and try again.';



      cameraError

        .classList

        .remove('hidden');



      scanningMessage

        .classList

        .add('hidden');


    }


  }




  // ======================================================

  // QR SUCCESS

  // ======================================================


  async function onScanSuccess(

    decodedText

  ) {



    if (scanLocked) {

      return;

    }



    scanLocked = true;



    scanningMessage

      .classList

      .add('hidden');



    verifyMessage

      .classList

      .remove('hidden');



    await stopScanner();



    // ----------------------------------------------------

    // BASIC IDC QR CHECK

    // Avoid sending random QR codes to backend

    // ----------------------------------------------------


    const qrValue =

      String(

        decodedText || ''

      ).trim();



    if (

      !qrValue.startsWith(

        'IDC2026|'

      )

    ) {



      cameraError

        .textContent =

        '⚠️ This is not an IDC Garba Raasverse 2026 entry pass.';



      cameraError

        .classList

        .remove('hidden');



      verifyMessage

        .classList

        .add('hidden');



      scanLocked = false;



      setTimeout(

        startScanner,

        1800

      );



      return;


    }



    // ----------------------------------------------------

    // SEND QR TO YOUR EXISTING APPS SCRIPT BACKEND

    // ----------------------------------------------------


    const destination =

      APPS_SCRIPT_URL +

      '?qr=' +

      encodeURIComponent(

        qrValue

      );



    window.location.href =

      destination;


  }




  // ======================================================

  // STOP CAMERA

  // ======================================================


  async function stopScanner() {



    if (!scanner) {

      return;

    }



    try {



      const state =

        scanner.getState();



      if (

        state ===

        Html5QrcodeScannerState.SCANNING ||

        state ===

        Html5QrcodeScannerState.PAUSED

      ) {


        await scanner.stop();


      }



      scanner.clear();



    }

    catch (error) {


      console.log(

        'Scanner stop:',

        error

      );


    }



    scanner = null;


  }




  // ======================================================

  // WHEN RETURNING FROM RESULT PAGE

  // ======================================================


  window.addEventListener(

    'pageshow',

    function() {



      scanLocked = false;



      verifyMessage

        .classList

        .add('hidden');



      scannerScreen

        .classList

        .add('hidden');



      welcomeScreen

        .classList

        .remove('hidden');


    }

  );
